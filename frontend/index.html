// Production-grade Singly Linked List Implementation
#include <iostream>
#include <string>
#include <iomanip>

class Employee {
public:
    int id;
    std::string name;
    std::string designation;
    double salary;
    Employee* next;

    Employee(int id, const std::string& name, const std::string& des, double sal)
        : id(id), name(name), designation(des), salary(sal), next(nullptr) {}
};

class Organization {
private:
    Employee* head;
    int count;

public:
    Organization() : head(nullptr), count(0) {}

    ~Organization() {
        clear();
    }

    void clear() {
        Employee* curr = head;
        while (curr != nullptr) {
            Employee* temp = curr;
            curr = curr->next;
            delete temp;
        }
        head = nullptr;
        count = 0;
    }

    bool insert(int id, const std::string& name, const std::string& des, double salary) {
        if (id <= 0 || exists(id)) return false;
        Employee* newNode = new Employee(id, name, des, salary);
        if (!head) {
            head = newNode;
        } else {
            Employee* curr = head;
            while (curr->next) curr = curr->next;
            curr->next = newNode;
        }
        count++;
        return true;
    }

    bool deleteById(int id) {
        if (!head) return false;
        if (head->id == id) {
            Employee* temp = head;
            head = head->next;
            delete temp;
            count--;
            return true;
        }
        Employee *curr = head, *prev = nullptr;
        while (curr && curr->id != id) {
            prev = curr;
            curr = curr->next;
        }
        if (!curr) return false;
        prev->next = curr->next;
        delete curr;
        count--;
        return true;
    }

    void sortById(bool asc = true) {
        if (!head || !head->next) return;
        for (Employee* p = head; p != nullptr; p = p->next) {
            for (Employee* q = p->next; q != nullptr; q = q->next) {
                if (asc ? (p->id > q->id) : (p->id < q->id)) {
                    std::swap(p->id, q->id);
                    std::swap(p->name, q->name);
                    std::swap(p->designation, q->designation);
                    std::swap(p->salary, q->salary);
                }
            }
        }
    }
};

College Viva & Technical Interview Q&A

1. Why use a Linked List instead of an Array for Employee Management?

Arrays have fixed sizes (or need expensive $O(N)$ reallocation). Linked Lists allow dynamic memory allocation on demand, meaning employees can be inserted or deleted without shifting memory blocks or pre-allocating large fixed buffers.

2. What is the time complexity of inserting a node at the head vs tail?

Inserting at the Head is $O(1)$ constant time (update pointer). Inserting at the Tail is $O(N)$ linear time in a Singly Linked List because you must traverse from head to the last node (unless a dedicated tail pointer is maintained, in which case it becomes $O(1)$).

3. Why is deleting the head node a special case?

Because there is no preceding node whose next pointer needs updating. Instead, the entry pointer head itself must be redirected to head->next before deallocating the original node.

4. How does Bubble Sort operate on a Singly Linked List?

Two pointers p and q = p->next traverse the list in nested loops ($O(N^2)$). When comparison conditions match, swapping the node payloads (data fields) avoids complex pointer relinking.
